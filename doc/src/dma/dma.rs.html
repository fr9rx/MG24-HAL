1//! LDMA (Linked Direct Memory Access) Controller
2//!
3//! Provides DMA support for memory-to-memory, memory-to-peripheral, and
4//! peripheral-to-memory transfers using the EFR32MG24's LDMA controller.
5//!
6//! # Example
7//!
8//! ```ignore
9//! use mg24_hal::{CpuConfig, dma::{Dma, DmaChannel, DmaConfig, DmaRequest}};
10//!
11//! let dp = mg24_hal::init(CpuConfig::default()).unwrap();
12//! let mut dma = Dma::new();
13//!
14//! // Configure channel 0 for I2C0 RX
15//! let config = DmaConfig::default()
16//!     .with_channel(0)
17//!     .with_request(DmaRequest::I2c0Rx);
18//!
19//! // Set up a transfer
20//! dma.configure_transfer(
21//!     0,
22//!     0x4000_a00C as u32,  // I2C0 RXDATA register
23//!     &mut buffer[0] as *mut _ as u32,
24//!     32,  // 32 bytes to transfer
25//! );
26//!
27//! dma.enable_channel(0);
28//! dma.start_transfer(0);
29//!
30//! while !dma.is_transfer_done(0) {}
31//! dma.clear_done_flag(0);
32//! ```
33

35/// DMA request sources for peripheral triggering
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37pub enum DmaRequest {
38    /// I2C0 receive FIFO data valid
39    I2c0Rx,
40    /// I2C0 transmit buffer level
41    I2c0Tx,
42    /// I2C1 receive FIFO data valid
43    I2c1Rx,
44    /// I2C1 transmit buffer level
45    I2c1Tx,
46}
47
48impl DmaRequest {
49    /// Get the SOURCESEL value for this request
50    fn sourcesel(&self) -> u32 {
51        match self {
52            DmaRequest::I2c0Rx | DmaRequest::I2c0Tx => 0x5,
53            DmaRequest::I2c1Rx | DmaRequest::I2c1Tx => 0x6,
54        }
55    }
56
57    /// Get the SIGSEL value for this request
58    fn sigsel(&self) -> u32 {
59        match self {
60            DmaRequest::I2c0Rx | DmaRequest::I2c1Rx => 0x0,
61            DmaRequest::I2c0Tx | DmaRequest::I2c1Tx => 0x1,
62        }
63    }
64}

66/// DMA transfer size configuration
67#[derive(Debug, Clone, Copy, PartialEq, Eq)]
68pub enum DmaSize {
69    /// Byte (8-bit) transfers
70    Byte = 0,
71    /// Half-word (16-bit) transfers
72    HalfWord = 1,
73    /// Word (32-bit) transfers
74    Word = 2,
75}
76
77/// DMA address increment configuration
78#[derive(Debug, Clone, Copy, PartialEq, Eq)]
79pub enum DmaIncrement {
80    /// Increment by 1 unit
81    One = 0,
82    /// Increment by 2 units
83    Two = 1,
84    /// Increment by 4 units
85    Four = 2,
86    /// No increment (fixed address)
87    None = 3,
88}

90/// DMA channel configuration builder
91#[derive(Debug, Clone)]
92pub struct DmaConfig {
93    channel: u8,
94    request: Option<DmaRequest>,
95    size: DmaSize,
96    src_inc: DmaIncrement,
97    dst_inc: DmaIncrement,
98    block_size: u16,
99}
100
101impl DmaConfig {
102    /// Set the DMA channel (0-7)
103    pub fn with_channel(mut self, ch: u8) -> Self {
104        self.channel = ch;
105        self
106    }
107
108    /// Set the DMA request source
109    pub fn with_request(mut self, request: DmaRequest) -> Self {
110        self.request = Some(request);
111        self
112    }
113
114    /// Set the transfer size
115    pub fn with_size(mut self, size: DmaSize) -> Self {
116        self.size = size;
117        self
118    }
119
120    /// Set the source address increment
121    pub fn with_src_inc(mut self, inc: DmaIncrement) -> Self {
122        self.src_inc = inc;
123        self
124    }
125
126    /// Set the destination address increment
127    pub fn with_dst_inc(mut self, inc: DmaIncrement) -> Self {
128        self.dst_inc = inc;
129        self
130    }
131
132    /// Set the block transfer size (units per arbitration cycle)
133    pub fn with_block_size(mut self, size: u16) -> Self {
134        self.block_size = size;
135        self
136    }
137}

152/// LDMA (Linked Direct Memory Access) Controller
153///
154/// Provides access to the EFR32MG24's LDMA controller for DMA transfers.
155/// The controller has 8 independent channels (0-7) that can be configured
156/// for various data transfer scenarios.
157pub struct Dma {
158    _private: (),
159}
160
161impl Dma {
162    /// Create a new LDMA controller instance and enable the module
163    ///
164    /// # Safety
165    ///
166    /// The LDMA controller must only be instantiated once. Multiple instances
167    /// will cause undefined behavior.
168    pub fn new() -> Self {
169        unsafe {
170            // Enable the LDMA module (RM 24.7.2 LDMA_EN)
171            let en_ptr = (0x4001_2000 + 0x004) as *mut u32;
172            core::ptr::write_volatile(en_ptr, 0x1);
173        }
174
175        Self { _private: () }
176    }
177
178    /// Configure channel peripheral request source
179    ///
180    /// This sets up the DMA request selection register to connect a channel
181    /// to a peripheral trigger source.
182    pub fn configure_request(&mut self, channel: u8, request: DmaRequest) {
183        assert!(channel < 8, "Channel must be 0-7");
184
185        unsafe {
186            // Calculate register offset for this channel
187            // LDMAXBAR_CHx_REQSEL is at 0x4001_2004 + (channel * 4)
188            let ptr = (0x4001_2004 + (channel as usize * 4)) as *mut u32;
189
190            // Write SOURCESEL and SIGSEL
191            // RM 24.9.2 LDMAXBAR_CHx_REQSEL
192            let reqsel = (request.sourcesel() << 16) | request.sigsel();
193            core::ptr::write_volatile(ptr, reqsel);
194        }
195    }
196
197    /// Configure the transfer for a channel
198    ///
199    /// Sets up source address, destination address, and transfer count.
200    /// The transfer is not started until `enable_channel()` and `start_transfer()` are called.
201    pub fn configure_transfer(
202        &mut self,
203        channel: u8,
204        src_addr: u32,
205        dst_addr: u32,
206        xfer_count: u16,
207        config: &DmaConfig,
208    ) {
209        assert!(channel < 8, "Channel must be 0-7");
210        assert!(xfer_count > 0, "Transfer count must be > 0");
211
212        unsafe {
213            // Calculate base offset for this channel's registers
214            // Channels are spaced 0x20 bytes apart (RM 24.6)
215            let ch_offset = channel as usize * 0x20;
216
217            // Set source address (offset 0x68 for CH0, +ch_offset)
218            let src_ptr = (0x4001_2000 + 0x68 + ch_offset) as *mut u32;
219            core::ptr::write_volatile(src_ptr, src_addr);
220
221            // Set destination address (offset 0x6C)
222            let dst_ptr = (0x4001_2000 + 0x6C + ch_offset) as *mut u32;
223            core::ptr::write_volatile(dst_ptr, dst_addr);
224
225            // Configure control register (offset 0x64)
226            // RM 24.7.25 LDMA_CHx_CTRL
227            let ctrl_ptr = (0x4001_2000 + 0x64 + ch_offset) as *mut u32;
228            let ctrl = (1 << 31) |  // DSTMODE: absolute (0)
229                       (0 << 30) |  // SRCMODE: absolute (0)
230                       ((config.dst_inc as u32) << 28) |  // DSTINC
231                       ((config.size as u32) << 26) |  // SIZE
232                       ((config.src_inc as u32) << 24) |  // SRCINC
233                       (0 << 23) |  // IGNORESREQ: 0
234                       (0 << 22) |  // DECLOOPCNT: 0
235                       (0 << 21) |  // REQMODE: BLOCK mode (0)
236                       (0 << 20) |  // DONEIEN: 0 (disable interrupt)
237                       ((block_size_bits(config.block_size) as u32) << 16) |  // BLOCKSIZE
238                       (0 << 15) |  // BYTESWAP: 0
239                       (((xfer_count - 1) as u32) << 4) |  // XFERCNT (one less than actual)
240                       (0 << 0);  // STRUCTTYPE: TRANSFER (0)
241
242            core::ptr::write_volatile(ctrl_ptr, ctrl);
243        }
244    }
245
246    /// Enable a DMA channel
247    ///
248    /// This marks the channel as ready to accept requests.
249    pub fn enable_channel(&mut self, channel: u8) {
250        assert!(channel < 8, "Channel must be 0-7");
251
252        unsafe {
253            // Write to LDMA_CHEN (offset 0x24)
254            // RM 24.7.10 LDMA_CHEN
255            let chen_ptr = (0x4001_2000 + 0x24) as *mut u32;
256            core::ptr::write_volatile(chen_ptr, 1 << channel);
257        }
258    }
259
260    /// Disable a DMA channel
261    ///
262    /// Clears any pending transfers on the channel.
263    pub fn disable_channel(&mut self, channel: u8) {
264        assert!(channel < 8, "Channel must be 0-7");
265
266        unsafe {
267            // Write to LDMA_CHDIS (offset 0x28)
268            // RM 24.7.11 LDMA_CHDIS
269            let chdis_ptr = (0x4001_2000 + 0x28) as *mut u32;
270            core::ptr::write_volatile(chdis_ptr, 1 << channel);
271        }
272    }
273
274    /// Start a software-triggered DMA transfer
275    ///
276    /// This initiates a transfer on the specified channel by writing to the
277    /// LDMA_SWREQ register.
278    pub fn start_transfer(&mut self, channel: u8) {
279        assert!(channel < 8, "Channel must be 0-7");
280
281        unsafe {
282            // Write to LDMA_SWREQ (offset 0x3C)
283            // RM 24.7.16 LDMA_SWREQ
284            let swreq_ptr = (0x4001_2000 + 0x3C) as *mut u32;
285            core::ptr::write_volatile(swreq_ptr, 1 << channel);
286        }
287    }
288
289    /// Check if a channel's transfer is complete
290    ///
291    /// Returns true if the CHDONE flag is set for this channel.
292    pub fn is_transfer_done(&self, channel: u8) -> bool {
293        assert!(channel < 8, "Channel must be 0-7");
294
295        unsafe {
296            // Read LDMA_CHDONE (offset 0x34)
297            // RM 24.7.14 LDMA_CHDONE
298            let chdone_ptr = (0x4001_2000 + 0x34) as *const u32;
299            let chdone = core::ptr::read_volatile(chdone_ptr);
300            (chdone & (1 << channel)) != 0
301        }
302    }
303
304    /// Clear the done flag for a channel
305    ///
306    /// This must be called after a transfer completes to clear the interrupt flag.
307    pub fn clear_done_flag(&mut self, channel: u8) {
308        assert!(channel < 8, "Channel must be 0-7");
309
310        unsafe {
311            // Write to LDMA_CHDONE (offset 0x34)
312            // RM 24.7.14 LDMA_CHDONE - writing 1 clears the bit
313            let chdone_ptr = (0x4001_2000 + 0x34) as *mut u32;
314            core::ptr::write_volatile(chdone_ptr, 1 << channel);
315        }
316    }
317
318    /// Check if any channel is currently busy
319    pub fn any_busy(&self) -> bool {
320        unsafe {
321            // Read LDMA_STATUS (offset 0x0C)
322            // RM 24.7.4 LDMA_STATUS - ANYBUSY is bit 0
323            let status_ptr = (0x4001_2000 + 0x0C) as *const u32;
324            let status = core::ptr::read_volatile(status_ptr);
325            (status & 0x1) != 0
326        }
327    }
328
329    /// Get the current status of all channels
330    pub fn channel_status(&self) -> u8 {
331        unsafe {
332            // Read LDMA_CHSTATUS (offset 0x2C)
333            // RM 24.7.12 LDMA_CHSTATUS
334            let status_ptr = (0x4001_2000 + 0x2C) as *const u32;
335            (core::ptr::read_volatile(status_ptr) & 0xFF) as u8
336        }
337    }
338}

340/// Convert block size to BLOCKSIZE register encoding
341/// RM 24.7.25 LDMA_CHx_CTRL - BLOCKSIZE field
342fn block_size_bits(size: u16) -> u8 {
343    match size {
344        1 => 0x0,
345        2 => 0x1,
346        3 => 0x2,
347        4 => 0x3,
348        6 => 0x4,
349        8 => 0x5,
350        16 => 0x7,
351        32 => 0x9,
352        64 => 0xA,
353        128 => 0xB,
354        256 => 0xC,
355        512 => 0xD,
356        1024 => 0xE,
357        _ => 0x0,
358    }
359}
